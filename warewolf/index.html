PHP & DB CONFIGURATION

This frontend connects to backend.php. It automatically falls back to client mock mode if PHP is unreachable. Below is the standard PHP script tailored for your werewolf_db and ../config.ini reader setup. 

<?php
// backend.php - Werewolf Game PHP API
header('Content-Type: application/json');

// Read database credentials from ../config.ini
// Row 1 = Username, Row 2 = Password
$iniPath = __DIR__ . '/../config.ini';
$dbUser = 'root';
$dbPass = '';

if (file_exists($iniPath)) {
    $lines = file($iniPath, FILE_IGNORE_NEW_LINES | FILE_SKIP_EMPTY_LINES);
    if (isset($lines[0])) $dbUser = trim($lines[0]);
    if (isset($lines[1])) $dbPass = trim($lines[1]);
}

$dbHost = 'localhost';
$dbName = 'werewolf_db';

try {
    $pdo = new PDO("mysql:host=$dbHost;dbname=$dbName;charset=utf8mb4", $dbUser, $dbPass, [
        PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC
    ]);
} catch (Exception $e) {
    echo json_encode(['status' => 'error', 'message' => 'DB Connection Failed: ' . $e->getMessage()]);
    exit;
}

$action = $_GET['action'] ?? '';

if ($action === 'create_room') {
    $roomCode = strtoupper(substr(md5(uniqid()), 0, 6));
    echo json_encode(['status' => 'success', 'roomCode' => $roomCode]);
} elseif ($action === 'join_room') {
    $code = $_GET['code'] ?? '';
    echo json_encode(['status' => 'success', 'roomCode' => $code]);
} else {
    echo json_encode(['status' => 'success', 'message' => 'Werewolf API Operational']);
}
?>
